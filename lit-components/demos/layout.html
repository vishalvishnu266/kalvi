Usage cheatsheet

<!-- Vertical rhythm -->
<ui-stack gap="md"> … </ui-stack>

<!-- Wrapping inline row (chips, meta, action bars) -->
<ui-cluster gap="sm" justify="between"> … </ui-cluster>

<!-- Auto-fit responsive tiles -->
<ui-grid min-col="240" gap="md"> … </ui-grid>

<!-- Fixed cols -->
<ui-grid cols="12" gap="sm"> … </ui-grid>

<!-- N-column with ratios -->
<ui-columns ratios="1 2 1" gap="md"> … </ui-columns>

<!-- Sidebar + main (mobile-stacks) -->
<ui-sidebar width="240px" collapse-at="768">
  <nav slot="side"> … </nav>
  <main> … </main>
</ui-sidebar>

<!-- Centered readable column -->
<ui-center max-w="72ch" padded> <article> … </article> </ui-center>
